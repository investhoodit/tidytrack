-- ✅ Create Database
CREATE DATABASE IF NOT EXISTS hood;
USE hood;

-- ✅ Users Table
CREATE TABLE users (
    userID INT AUTO_INCREMENT PRIMARY KEY,
    email VARCHAR(100) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    role ENUM('manager','member') DEFAULT 'member',
    team_id INT DEFAULT NULL,
    FOREIGN KEY (team_id) REFERENCES teams(id) ON DELETE SET NULL
);

-- ✅ Teams Table
CREATE TABLE teams (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL
);

-- ✅ Team Members Table (Optional, if you store members separately)
CREATE TABLE team_members (
    id INT AUTO_INCREMENT PRIMARY KEY,
    team_id INT NOT NULL,
    name VARCHAR(100) NOT NULL,
    FOREIGN KEY (team_id) REFERENCES teams(id) ON DELETE CASCADE
);

-- ✅ Projects Table
CREATE TABLE projects (
    id INT AUTO_INCREMENT PRIMARY KEY,
    team_id INT NOT NULL,
    name VARCHAR(150) NOT NULL,
    description TEXT,
    FOREIGN KEY (team_id) REFERENCES teams(id) ON DELETE CASCADE
);

-- ✅ Tasks Table
CREATE TABLE tasks (
    taskID INT AUTO_INCREMENT PRIMARY KEY,
    taskTitle VARCHAR(150) NOT NULL,
    description TEXT,
    createdBy INT NOT NULL,
    team_id INT,
    assignTo INT,
    priority ENUM('low','medium','high') DEFAULT 'medium',
    dueDate DATE,
    FOREIGN KEY (team_id) REFERENCES teams(id) ON DELETE CASCADE,
    FOREIGN KEY (assignTo) REFERENCES users(userID) ON DELETE SET NULL
);

-- ✅ Timesheets Table
CREATE TABLE timesheets (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    date DATE NOT NULL,
    hours_worked DECIMAL(5,2) NOT NULL,
    notes VARCHAR(255),
    task_id INT DEFAULT NULL,
    FOREIGN KEY (user_id) REFERENCES users(userID) ON DELETE CASCADE,
    FOREIGN KEY (task_id) REFERENCES tasks(taskID) ON DELETE SET NULL
);
